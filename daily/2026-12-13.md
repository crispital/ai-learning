```python
import argparse           #用于读取命令行参数
import json               #用于保存json文件
import random             #Python自带的随机数
import sys                #访问 Python 运行环境和模块搜索路径
from pathlib import Path  #处理文件和文件夹路径

import numpy as np        #保存数组，设置随机种子
import torch              
from torch import nn      #nn（Neutral Net），里面还有我们需要的损失函数  
from torch.utils.data import DataLoader      #torch.utils.data：是一个数据加载工具；DataLoader：用于把数据集分成一个个batch
from torchvision import datasets, transforms
#datasets：提供MNIST数据集（还有很多预设好的数据类别）；transforms：对图片进行预处理


PROJECT_ROOT = Path(__file__).resolve().parents[1]
# 获取当前项目的根目录。注意parents[1]是根目录，parents[0]是训练脚本所在目录
if str(PROJECT_ROOT) not in sys.path:
    sys.path.insert(0, str(PROJECT_ROOT))  # 支持直接执行本脚本时导入src。

from src.mnist_model import MnistCNN


MNIST_MEAN = 0.1307  # MNIST归一化均值。
MNIST_STD = 0.3081   # MNIST归一化标准差。
#MNIST图片转化为张量后，像素通常都位于0-1，因为神经网络基本只支持0-1的参数输入（从uint8到float32）归一化像素 = (原始像素 - 均值) / 标准差；归一化后往往能让网络训练的更加稳定。

#接下来固定随机种子，确定以后能复现。这里有个细节，seed=42是默认参数，但是这三种随机数生成器生成对同一种子的生成结果是不一样的。
def set_random_seed(seed):
    """固定Python、NumPy和PyTorch的随机种子。"""
    random.seed(seed)
    np.random.seed(seed)
    torch.manual_seed(seed)

#这个函数是不参与训练的，而是用于把一组固定测试图片保存了。
def save_fixed_samples(inputs, labels, output_path):
    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)
#这里解释起来很复杂，但说到底就是为了先确认一个保存路径，如果这个保存路径存在就跳过，如果不存在就创建这个路径。

    normalized_inputs = inputs.detach().cpu().to(torch.float32)
    # detach()：让这些数据脱离梯度计算；cpu()：把数据放到 CPU；to(torch.float32)：确保类型为 float32
    #这里保存的 normalized_inputs 是模型真正接收到的数据，也就是归一化之后的输入。
    images = (normalized_inputs * MNIST_STD + MNIST_MEAN).clamp(0.0, 1.0)
    # 反归一化，使得图片回到原来的（0-1），.clamp()确保数值范围。
    np.savez_compressed(
        output_path,
        inputs=normalized_inputs.numpy(),
        images=images.numpy(),
        labels=labels.detach().cpu().to(torch.int64).numpy(),
    )
#这是为了保存三个数组，

def train_one_epoch(model, data_loader, optimizer, loss_function, device):
    """完成一轮参数更新并返回按样本数计算的平均损失。"""
    model.train()  # 显式切换为训练状态。
    total_loss = 0.0
    total_samples = 0

    for inputs, labels in data_loader:
        inputs = inputs.to(device)
        labels = labels.to(device)

        optimizer.zero_grad()  # 清空上一批次留下的梯度。
        logits = model(inputs)
        loss = loss_function(logits, labels)
        loss.backward()
        optimizer.step()

        batch_size = labels.size(0)  # 按实际样本数计算整轮平均损失。
        total_loss += loss.item() * batch_size
        total_samples += batch_size

    return total_loss / total_samples


def evaluate(model, data_loader, loss_function, device):
    """在不更新参数的情况下计算测试集损失和分类准确率。"""
    model.eval()
    total_loss = 0.0
    total_correct = 0
    total_samples = 0

    with torch.inference_mode():  # 评估时不记录梯度。
        for inputs, labels in data_loader:
            inputs = inputs.to(device)
            labels = labels.to(device)
            logits = model(inputs)
            loss = loss_function(logits, labels)

            batch_size = labels.size(0)
            total_loss += loss.item() * batch_size
            total_correct += (logits.argmax(dim=1) == labels).sum().item()
            total_samples += batch_size

    return {
        "loss": total_loss / total_samples,
        "accuracy": total_correct / total_samples,
        "sample_count": total_samples,
    }


def collect_fixed_samples(dataset, sample_count):
    """按固定顺序取测试集开头若干样本，作为跨运行时共同输入。"""
    samples = [dataset[index] for index in range(sample_count)]
    inputs = torch.stack([sample[0] for sample in samples])
    labels = torch.tensor([sample[1] for sample in samples], dtype=torch.int64)
    return inputs, labels


def write_json(data, output_path):
    """使用UTF-8写入便于人工检查和版本管理的JSON文件。"""
    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    output_path.write_text(
        json.dumps(data, ensure_ascii=False, indent=2),
        encoding="utf-8",
    )


def parse_args():
    """读取训练参数；默认值对应本章的基础CPU实验。"""
    parser = argparse.ArgumentParser(description="在CPU上训练MNIST简单卷积神经网络")
    parser.add_argument("--data-dir", type=Path, default=PROJECT_ROOT / "data")
    parser.add_argument("--epochs", type=int, default=3)
    parser.add_argument("--batch-size", type=int, default=64)
    parser.add_argument("--learning-rate", type=float, default=1e-3)
    parser.add_argument("--seed", type=int, default=42)
    parser.add_argument("--fixed-sample-count", type=int, default=10)
    return parser.parse_args()


def main():
    args = parse_args()
    set_random_seed(args.seed)
    device = torch.device("cpu")  # 本章使用CPU建立正确性基线。

    transform = transforms.Compose(  # 先转为[0,1]张量，再执行归一化。
        [
            transforms.ToTensor(),
            transforms.Normalize((MNIST_MEAN,), (MNIST_STD,)),
        ]
    )
    train_dataset = datasets.MNIST(
        root=args.data_dir,
        train=True,
        transform=transform,
        download=True,
    )
    test_dataset = datasets.MNIST(
        root=args.data_dir,
        train=False,
        transform=transform,
        download=True,
    )

    loader_generator = torch.Generator().manual_seed(args.seed)  # 固定打乱顺序。
    train_loader = DataLoader(
        train_dataset,
        batch_size=args.batch_size,
        shuffle=True,
        num_workers=0,  # 单进程读取，减少入门环境中的额外变量。
        generator=loader_generator,
    )
    test_loader = DataLoader(
        test_dataset,
        batch_size=args.batch_size,
        shuffle=False,
        num_workers=0,
    )

    model = MnistCNN().to(device)
    loss_function = nn.CrossEntropyLoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=args.learning_rate)

    epoch_records = []
    for epoch in range(1, args.epochs + 1):
        train_loss = train_one_epoch(
            model,
            train_loader,
            optimizer,
            loss_function,
            device,
        )
        evaluation = evaluate(model, test_loader, loss_function, device)
        epoch_record = {
            "epoch": epoch,
            "train_loss": train_loss,
            "test_loss": evaluation["loss"],
            "test_accuracy": evaluation["accuracy"],
        }
        epoch_records.append(epoch_record)
        print(
            f"Epoch {epoch}/{args.epochs} "
            f"train_loss={train_loss:.6f} "
            f"test_loss={evaluation['loss']:.6f} "
            f"test_accuracy={evaluation['accuracy']:.4%}"
        )

    weights_path = PROJECT_ROOT / "models" / "pytorch" / "mnist_cnn_state_dict.pt"
    weights_path.parent.mkdir(parents=True, exist_ok=True)
    torch.save(model.state_dict(), weights_path)  # 仅保存模型参数。

    config = {  # 保存导出和推理共同使用的模型接口。
        "model_name": "mnist_cnn",
        "model_class": "src.mnist_model.MnistCNN",
        "input_name": "input",
        "input_shape": [1, 1, 28, 28],
        "input_dtype": "float32",
        "input_layout": "NCHW",
        "normalization": {"mean": [MNIST_MEAN], "std": [MNIST_STD]},
        "output_name": "logits",
        "output_shape": [1, 10],
        "classes": list(range(10)),
        "seed": args.seed,
    }
    config_path = PROJECT_ROOT / "models" / "pytorch" / "mnist_cnn_config.json"
    write_json(config, config_path)

    fixed_inputs, fixed_labels = collect_fixed_samples(
        test_dataset,
        args.fixed_sample_count,
    )
    samples_path = PROJECT_ROOT / "samples" / "mnist_fixed_samples.npz"
    save_fixed_samples(fixed_inputs, fixed_labels, samples_path)  # 固定对比输入。

    report = {  # 独立保存训练结果和关键环境版本。
        "environment": {
            "python": sys.version.split()[0],
            "torch": torch.__version__,
            "torchvision": __import__("torchvision").__version__,
            "numpy": np.__version__,
        },
        "training": {
            "device": str(device),
            "epochs": args.epochs,
            "batch_size": args.batch_size,
            "learning_rate": args.learning_rate,
            "seed": args.seed,
        },
        "epochs": epoch_records,
        "artifacts": {
            "weights": str(weights_path.relative_to(PROJECT_ROOT)),
            "config": str(config_path.relative_to(PROJECT_ROOT)),
            "fixed_samples": str(samples_path.relative_to(PROJECT_ROOT)),
        },
    }
    report_path = PROJECT_ROOT / "reports" / "training_result.json"
    write_json(report, report_path)

    print(f"权重：{weights_path}")
    print(f"模型配置：{config_path}")
    print(f"固定样本：{samples_path}")
    print(f"训练记录：{report_path}")


if __name__ == "__main__":
    main()
``` 
